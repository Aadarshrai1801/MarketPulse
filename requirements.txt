# --- Data storage ---
pymongo[srv]==4.17.0       # MongoDB driver (users + product price history); [srv] enables mongodb+srv:// URIs

# --- Scraper (Scrapling, no browser needed on free Render) ---
# Fetcher (plain HTTP + Chrome TLS impersonation) is the default path and
# needs only this package - no `scrapling install` browser download, so the
# image stays slim (~200MB) and fits Render free (512MB RAM).
# StealthyFetcher (headless Chromium, Cloudflare Turnstile bypass) is used
# ONLY as a blocked-page fallback for fetch_mode="auto" retailers (see
# scraper/config.py). It needs browsers from `scrapling install`, which is
# intentionally NOT run in the Dockerfile; without them the row just fails
# with a clear error instead of crashing the job.
scrapling[fetchers]==0.4.15

# --- OCR endpoint (/api/ocr, optional on free Render) ---
# Only the light image-decode deps are installed here so /api/ocr can still
# validate uploads and return a clear "engine not installed" 503.
# The heavy PaddleOCR engine (paddlepaddle==2.6.2 + paddleocr==2.7.3, ~1GB,
# needs numpy<2) is intentionally NOT installed - it OOMs Render free and
# blows the build time. To re-enable OCR on a bigger host:
#   pip install "paddlepaddle==2.6.2" "paddleocr==2.7.3" "numpy==1.26.4"
# (keep numpy pinned <2.0 while paddle is present; see old comment history).
opencv-python-headless==4.11.0.86   # headless build - no GUI deps needed on a server
numpy==1.26.4
pandas==2.2.2

# --- Production server ---
gunicorn==26.0.0           # run with: gunicorn -w 1 -b 0.0.0.0:$PORT app:app
                           # NOTE: -w 1, not -w 2 - see "Single worker" note below
                           # (Windows/dev alternative: waitress)

# --- Worker count note ---
# app.py's JOBS dict is in-memory (see README "Single worker" section).
# With gunicorn -w 2+, a job started on one worker won't be visible when
# another worker handles the /api/jobs/<id> poll request - random 404s/stuck
# "running" state. Keep -w 1 unless JOBS is moved to Redis/a DB.
